// ===================== STEP INSTRUCTIONS WIDGET =====================
function StepInstructionsWidget() {
  const [mode, setMode] = useState('single'); // 'single' | 'all'
  const [edit, setEdit] = useState(false);
  const [hideText, setHideText] = useState(false); // studentvy: döljer brödtext
  const [current, setCurrent] = useState(0);
  const [picker, setPicker] = useState({ open: false, stepIndex: 0 });

  const [steps, setSteps] = useState([
    {
      title: 'Läs uppgiften',
      body: 'Läs igenom uppgiften noggrant. Markera nyckelord och säkerställ att du förstår målet.',
      image: 'https://images.unsplash.com/photo-1513475382585-d06e58bcb0ea?q=80&w=800&auto=format&fit=crop',
      alt: 'Elev som läser en uppgift',
    },
    {
      title: 'Planera',
      body: 'Gör en snabb plan: Vad behöver göras först, sedan, sist? Vem gör vad och när?',
      image: 'https://images.unsplash.com/photo-1517245386807-bb43f82c33c4?q=80&w=800&auto=format&fit=crop',
      alt: 'Planering med post-its',
    },
    {
      title: 'Genomför',
      body: 'Jobba enligt planen. Stäm av halvvägs och justera vid behov. Lämna in när du är klar.',
      image: 'https://images.unsplash.com/photo-1486312338219-ce68d2c6f44d?q=80&w=800&auto=format&fit=crop',
      alt: 'Elev som arbetar vid laptop',
    },
  ]);

  function updateStep(i, patch) {
    setSteps((arr) => arr.map((s, idx) => (idx === i ? { ...s, ...patch } : s)));
  }

  const total = steps.length;
  const pct = ((current + 1) / total) * 100;

  return (
    <Card className="w-[min(980px,94vw)]">
      {/* Header */}
      <div className="px-5 pt-4 pb-2">
        <div className="flex items-center justify-between gap-3">
          <div className="flex items-center gap-2">
            <div className="h-7 w-7 grid place-items-center rounded-lg bg-white/70 border border-white/40">
              <NotebookPen className="h-4 w-4" />
            </div>
            <h2 className="text-sm font-medium tracking-wide uppercase text-slate-700">Stegvisa instruktioner</h2>
          </div>
          <div className="flex items-center gap-2">
            {/* mode switch */}
            <div className="bg-white/70 border border-white/40 rounded-xl p-1 flex text-sm">
              <button onClick={() => setMode('single')} className={`px-3 py-1.5 rounded-lg inline-flex items-center gap-1 ${mode==='single'?'bg-indigo-600 text-white':'text-slate-700 hover:bg-white'}`}><Columns3 className="h-4 w-4"/>Ett i taget</button>
              <button onClick={() => setMode('all')} className={`px-3 py-1.5 rounded-lg inline-flex items-center gap-1 ${mode==='all'?'bg-indigo-600 text-white':'text-slate-700 hover:bg-white'}`}><LayoutGrid className="h-4 w-4"/>Visa alla</button>
            </div>
            {/* edit + student view */}
            <button onClick={() => setEdit((v)=>!v)} className={`h-9 px-3 rounded-xl inline-flex items-center gap-2 border border-white/40 ${edit?'bg-black text-white':'bg-white/80 hover:bg-white'}` }><Edit3 className="h-4 w-4"/>Redigera</button>
            <button onClick={() => setHideText((v)=>!v)} className="h-9 px-3 rounded-xl bg-white/80 hover:bg-white border border-white/40 inline-flex items-center gap-2"><Eye className={`h-4 w-4 ${hideText?'hidden':''}`}/><EyeOff className={`h-4 w-4 ${hideText?'':'hidden'}`}/> {hideText? 'Visa text' : 'Dölj text'}</button>
          </div>
        </div>
      </div>

      {/* Progress for single mode */}
      {mode==='single' && (
        <div className="px-6">
          <div className="flex items-center justify-between text-xs text-slate-500 mb-2">
            <span>Steg {current+1} av {total}</span>
            <span>{Math.round(pct)}%</span>
          </div>
          <div className="h-2 w-full rounded-full bg-slate-200 overflow-hidden">
            <div className="h-full bg-gradient-to-r from-indigo-500 to-cyan-400" style={{width: pct+"%"}}/>
          </div>
        </div>
      )}

      {/* Body */}
      <div className="p-6 grid gap-6">
        {mode==='single' ? (
          <StepView
            step={steps[current]}
            index={current}
            total={total}
            edit={edit}
            hideText={hideText}
            onPrev={() => setCurrent((i)=> Math.max(0, i-1))}
            onNext={() => setCurrent((i)=> Math.min(total-1, i+1))}
            onChange={(patch)=> updateStep(current, patch)}
            onPickImage={() => setPicker({ open:true, stepIndex: current })}
          />
        ) : (
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
            {steps.map((s, i)=> (
              <StepCard
                key={i}
                step={s}
                index={i}
                hideText={hideText}
                edit={edit}
                onEdit={() => { setMode('single'); setCurrent(i); setEdit(true); }}
                onPickImage={() => { setMode('single'); setCurrent(i); setPicker({ open:true, stepIndex: i }); }}
              />
            ))}
          </div>
        )}
      </div>

      {/* Media Picker (mock) */}
      {picker.open && (
        <MediaPicker
          onClose={() => setPicker({ open:false, stepIndex:0 })}
          onSelect={(url)=> { updateStep(picker.stepIndex, { image: url }); setPicker({ open:false, stepIndex:0 }); }}
        />
      )}
    </Card>
  );
}

function StepView({ step, index, total, edit, hideText, onPrev, onNext, onChange, onPickImage }) {
  return (
    <div className="grid md:grid-cols-[1.2fr,1fr] gap-6 items-start">
      {/* Image */}
      <div className="relative rounded-2xl overflow-hidden ring-1 ring-black/10 bg-slate-100 min-h-[220px]">
        <img src={step.image} alt={step.alt || step.title} className="w-full h-full object-cover"/>
        <button onClick={onPickImage} className="absolute top-3 right-3 h-9 px-3 rounded-xl bg-white/90 hover:bg-white shadow inline-flex items-center gap-2">
          <ImageIcon className="h-4 w-4"/> Byt bild
        </button>
      </div>

      {/* Text */}
      <div>
        <div className="rounded-2xl p-5 bg-gradient-to-br from-indigo-600 to-purple-600 text-white shadow">
          {edit ? (
            <input
              value={step.title}
              onChange={(e)=> onChange({ title: e.target.value })}
              className="w-full bg-transparent text-2xl font-semibold outline-none"
            />
          ) : (
            <h3 className="text-2xl font-semibold">{step.title}</h3>
          )}
          <div className="mt-1 text-xs text-white/75">Steg {index+1} av {total}</div>
        </div>
        {!hideText && (
          <div className="mt-4">
            {edit ? (
              <textarea
                value={step.body}
                onChange={(e)=> onChange({ body: e.target.value })}
                rows={5}
                className="w-full rounded-xl border border-black/10 p-3 bg-white/80 focus:ring-2 focus:ring-indigo-500 outline-none"
              />
            ) : (
              <p className="text-slate-800 dark:text-slate-100 leading-relaxed bg-white/70 rounded-xl p-4">
                {step.body}
              </p>
            )}
          </div>
        )}

        {/* Nav */}
        <div className="mt-4 flex items-center gap-3">
          <button onClick={onPrev} className="inline-flex items-center gap-1 rounded-xl px-4 py-2 bg-white/80 hover:bg-white border border-white/40"><ChevronLeft className="h-4 w-4"/>Föregående</button>
          <button onClick={onNext} className="inline-flex items-center gap-1 rounded-xl px-4 py-2 bg-indigo-600 hover:bg-indigo-500 text-white"><ChevronRight className="h-4 w-4"/>Nästa</button>
        </div>
      </div>
    </div>
  );
}

function StepCard({ step, index, hideText, edit, onEdit, onPickImage }) {
  return (
    <div className="rounded-2xl overflow-hidden ring-1 ring-black/10 bg-white/70 hover:bg-white transition">
      <div className="relative h-36 bg-slate-100">
        <img src={step.image} alt={step.alt || step.title} className="w-full h-full object-cover"/>
        <div className="absolute top-2 left-2 text-xs bg-black/60 text-white px-2 py-1 rounded-full">Steg {index+1}</div>
        {edit && (
          <button onClick={onPickImage} className="absolute top-2 right-2 h-8 px-3 rounded-xl bg-white/90 hover:bg-white shadow inline-flex items-center gap-2">
            <ImageIcon className="h-4 w-4"/>Byt bild
          </button>
        )}
      </div>
      <div className="p-3">
        <div className="font-semibold text-slate-900">{step.title}</div>
        {!hideText && <div className="text-sm text-slate-600 mt-1 line-clamp-3">{step.body}</div>}
        <div className="mt-3">
          <button onClick={onEdit} className="h-8 px-3 rounded-lg bg-indigo-600 hover:bg-indigo-500 text-white text-sm">Öppna</button>
        </div>
      </div>
    </div>
  );
}

function MediaPicker({ onClose, onSelect }) {
  const [tab, setTab] = useState('pictos'); // 'photos' | 'pictos'
  const [q, setQ] = useState('');

  // Mock thumbnails
  const thumbs = Array.from({length: 12}).map((_,i)=>
    tab==='photos'
      ? `https://images.unsplash.com/photo-${1513475382585 + i}?q=80&w=600&auto=format&fit=crop`
      : `https://raw.githubusercontent.com/ARASAAC/arasaac-icons/master/PNG/512/animals/animal_${(i%8)+1}.png`
  );

  return (
    <div className="fixed inset-0 z-50">
      <div className="absolute inset-0 bg-slate-900/50 backdrop-blur-sm" onClick={onClose}/>
      <div className="absolute left-1/2 -translate-x-1/2 top-[6vh] w-[min(980px,94vw)] rounded-2xl bg-white shadow-2xl border border-black/10 overflow-hidden">
        <div className="flex items-center justify-between p-4 border-b border-black/10">
          <h3 className="text-lg font-semibold">Välj symbol</h3>
          <button onClick={onClose} className="h-9 px-3 rounded-xl bg-slate-100 hover:bg-slate-200">Stäng</button>
        </div>
        <div className="p-4">
          {/* Tabs */}
          <div className="flex gap-2">
            <button onClick={()=>setTab('photos')} className={`h-10 px-3 rounded-xl border ${tab==='photos'?'bg-indigo-600 text-white border-indigo-600':'bg-white border-black/10'}`}><ImageIcon className="h-4 w-4 inline mr-2"/>Foton (Pexels)</button>
            <button onClick={()=>setTab('pictos')} className={`h-10 px-3 rounded-xl border ${tab==='pictos'?'bg-indigo-600 text-white border-indigo-600':'bg-white border-black/10'}`}>🧩 Piktogram (ARASAAC)</button>
          </div>
          {/* Search */}
          <div className="mt-3 relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-500"/>
            <input value={q} onChange={(e)=>setQ(e.target.value)} placeholder={tab==='photos'? 'Search photos on Pexels…':'Search pictograms…'} className="w-full h-11 pl-9 pr-3 rounded-xl border border-black/10"/>
          </div>
          {/* Grid */}
          <div className="mt-4 grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3 max-h-[50vh] overflow-auto pr-1">
            {thumbs.map((src, i)=> (
              <div key={i} className="rounded-xl overflow-hidden ring-1 ring-black/10 bg-slate-100">
                <img src={src} alt="thumb" className="w-full h-36 object-cover"/>
                <div className="p-2 text-right">
                  <button onClick={()=>onSelect(src)} className="h-8 px-3 rounded-lg bg-indigo-600 hover:bg-indigo-500 text-white text-sm">Välj</button>
                </div>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}
