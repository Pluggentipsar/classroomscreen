// ===================== SHARED =====================

function Toggle({ checked, onChange, label, icon }) {
  return (
    <label className="inline-flex items-center gap-2 cursor-pointer">
      <button
        type="button"
        role="switch"
        aria-checked={checked}
        onClick={() => onChange(!checked)}
        className={`relative h-6 w-11 rounded-full transition-colors ${
          checked ? "bg-indigo-600" : "bg-slate-300 dark:bg-slate-600"
        }`}
        aria-label={label}
      >
        <span
          className={`absolute top-0.5 left-0.5 h-5 w-5 rounded-full bg-white shadow transition-transform ${
            checked ? "translate-x-5" : "translate-x-0"
          }`}
        />
      </button>
      {icon}
      <span className="text-sm text-slate-700/90 dark:text-slate-200/90">{label}</span>
    </label>
  );
}

function BigDigits({ seconds }) {
  // always show mm:ss in tabular figures
  const m = Math.floor(seconds / 60)
    .toString()
    .padStart(2, "0");
  const s = Math.floor(seconds % 60)
    .toString()
    .padStart(2, "0");
  return (
    <div className="font-semibold tabular-nums tracking-tight text-6xl md:text-7xl lg:text-8xl text-slate-900 dark:text-white">
      {m}:{s}
    </div>
  );
}

function RingProgress({ value, max, warn = false }) {
  const radius = 64; // px
  const circumference = 2 * Math.PI * radius;
  const pct = Math.max(0, Math.min(1, value / max));
  const offset = circumference * (1 - pct);
  const warnPulse = warn ? "animate-[pulse_1.1s_ease-in-out_infinite]" : "";

  return (
    <svg width={160} height={160} className="block">
      <defs>
        <linearGradient id="grad" x1="0%" y1="0%" x2="100%" y2="0%">
          <stop offset="0%" stopColor="#6366F1" />
          <stop offset="100%" stopColor="#22D3EE" />
        </linearGradient>
      </defs>
      <circle
        cx={80}
        cy={80}
        r={radius}
        stroke="currentColor"
        className="text-slate-300/80 dark:text-slate-700/70"
        strokeWidth={10}
        fill="none"
      />
      <circle
        cx={80}
        cy={80}
        r={radius}
        stroke="url(#grad)"
        strokeWidth={12}
        strokeLinecap="round"
        strokeDasharray={circumference}
        strokeDashoffset={offset}
        fill="none"
        className={`drop-shadow ${warnPulse}`}
      />
    </svg>
  );
}

function ActionButton({ onClick, icon, label, variant = "primary" }) {
  const variants = {
    primary:
      "bg-indigo-600 hover:bg-indigo-500 text-white shadow disabled:opacity-50",
    ghost:
      "bg-white/70 dark:bg-white/10 hover:bg-white/90 dark:hover:bg-white/20 text-slate-800 dark:text-slate-100 border border-white/40 dark:border-white/10",
  };
  return (
    <button
      onClick={onClick}
      className={`inline-flex items-center gap-2 rounded-xl px-4 py-2 text-sm font-medium transition ${variants[variant]}`}
    >
      {icon}
      <span>{label}</span>
    </button>
  );
}

// ===================== TIMER WIDGET =====================
function TimerWidget() {
  const [lengthMin, setLengthMin] = useState(5);
  const [secondsLeft, setSecondsLeft] = useState(lengthMin * 60);
  const [running, setRunning] = useState(false);
  const [soundOn, setSoundOn] = useState(true);
  const [pulseOn, setPulseOn] = useState(true);
  const [highContrast, setHighContrast] = useState(false);

  const maxSeconds = useMemo(() => lengthMin * 60, [lengthMin]);

  // Tick
  useEffect(() => {
    if (!running) return;
    const id = setInterval(() => {
      setSecondsLeft((s) => Math.max(0, s - 1));
    }, 1000);
    return () => clearInterval(id);
  }, [running]);

  // Reset seconds when length changes (only if not running)
  useEffect(() => {
    if (!running) setSecondsLeft(lengthMin * 60);
  }, [lengthMin, running]);

  // Simple beep when finished
  useEffect(() => {
    if (secondsLeft !== 0) return;
    if (!soundOn) return;
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const o = ctx.createOscillator();
    const g = ctx.createGain();
    o.type = "sine";
    o.frequency.value = 880;
    o.connect(g);
    g.connect(ctx.destination);
    g.gain.setValueAtTime(0.0001, ctx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.2, ctx.currentTime + 0.02);
    o.start();
    setTimeout(() => {
      g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.4);
      o.stop(ctx.currentTime + 0.45);
    }, 200);
  }, [secondsLeft, soundOn]);

  const warn = pulseOn && secondsLeft <= 10 && secondsLeft > 0;

  return (
    <Card
      className={
        highContrast
          ? "ring-2 ring-black bg-white text-black"
          : ""
      }
    >
      <Header title="Timer" />

      <div className="grid md:grid-cols-[160px,1fr] gap-6 p-6 pt-2">
        {/* Progress ring */}
        <div className="relative mx-auto">
          <RingProgress value={secondsLeft} max={maxSeconds} warn={warn} />
          <div className={`absolute inset-0 grid place-items-center ${warn ? "animate-pulse" : ""}`}>
            <BigDigits seconds={secondsLeft} />
          </div>
        </div>

        {/* Controls */}
        <div className="flex flex-col gap-4">
          {/* Length control */}
          <div className="flex items-center justify-between gap-3">
            <div className="text-xs uppercase tracking-wide text-slate-600 dark:text-slate-300">Längd</div>
            <div className="flex items-center gap-2">
              <button
                className="h-9 w-9 rounded-xl grid place-items-center bg-white/70 dark:bg-white/10 border border-white/40 dark:border-white/10 hover:bg-white/90 dark:hover:bg-white/20"
                onClick={() => setLengthMin((m) => Math.max(1, m - 1))}
                aria-label="Minska minuter"
              >
                –
              </button>
              <div className="min-w-[4ch] text-center font-semibold text-slate-900 dark:text-white">
                {lengthMin} min
              </div>
              <button
                className="h-9 w-9 rounded-xl grid place-items-center bg-white/70 dark:bg-white/10 border border-white/40 dark:border-white/10 hover:bg-white/90 dark:hover:bg-white/20"
                onClick={() => setLengthMin((m) => Math.min(99, m + 1))}
                aria-label="Öka minuter"
              >
                +
              </button>
            </div>
          </div>

          {/* Actions */}
          <div className="flex flex-wrap items-center gap-3">
            {running ? (
              <ActionButton
                onClick={() => setRunning(false)}
                icon={<Pause className="h-4 w-4" />}
                label="Pausa"
              />
            ) : (
              <ActionButton
                onClick={() => {
                  if (secondsLeft === 0) setSecondsLeft(maxSeconds);
                  setRunning(true);
                }}
                icon={<Play className="h-4 w-4" />}
                label="Starta"
              />
            )}
            <ActionButton
              variant="ghost"
              onClick={() => {
                setRunning(false);
                setSecondsLeft(maxSeconds);
              }}
              icon={<RotateCcw className="h-4 w-4" />}
              label="Återställ"
            />
          </div>

          {/* Toggles */}
          <div className="flex flex-wrap items-center gap-6">
            <Toggle
              checked={soundOn}
              onChange={setSoundOn}
              label={soundOn ? "Ljud på" : "Ljud av"}
              icon={soundOn ? <Volume2 className="h-4 w-4" /> : <VolumeX className="h-4 w-4" />}
            />
            <Toggle
              checked={pulseOn}
              onChange={setPulseOn}
              label={pulseOn ? "Visuell puls" : "Ingen puls"}
              icon={pulseOn ? <Eye className="h-4 w-4" /> : <EyeOff className="h-4 w-4" />}
            />
            <label className="inline-flex items-center gap-2 cursor-pointer">
              <button
                type="button"
                onClick={() => setHighContrast((v) => !v)}
                className={`inline-flex items-center gap-2 rounded-xl px-3 py-2 text-xs font-medium transition ${
                  highContrast
                    ? "bg-black text-white hover:bg-gray-900"
                    : "bg-white/70 dark:bg-white/10 hover:bg-white/90 dark:hover:bg-white/20 text-slate-800 dark:text-slate-100 border border-white/40 dark:border-white/10"
                }`}
                aria-label="High contrast"
              >
                <Contrast className="h-4 w-4" /> High Contrast
              </button>
            </label>
          </div>

          {/* Helper text */}
          <p className="text-xs text-slate-600/90 dark:text-slate-300/90">
            Tips: Kör i projector‑läge med mörk bakgrund och aktivera "High Contrast" i ljusa klassrum.
          </p>
        </div>
      </div>
    </Card>
  );
}

// ===================== SMART DOCK =====================
function SmartDock() {
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState("");
  const inputRef = useRef(null);
  const [favorites, setFavorites] = useState(["Background", "Poll", "Randomizer", "Music", "Timer"]);
  const [recent, setRecent] = useState([]);

  const widgets = [
    { id: "Background", label: "Background", icon: <ImageIcon className="h-5 w-5" /> },
    { id: "Poll", label: "Poll", icon: <BarChart3 className="h-5 w-5" /> },
    { id: "Randomizer", label: "Randomizer", icon: <Dice5 className="h-5 w-5" /> },
    { id: "Sound", label: "Sound level", icon: <Volume1 className="h-5 w-5" /> },
    { id: "Music", label: "Music", icon: <Music className="h-5 w-5" /> },
    { id: "Image", label: "Image", icon: <ImageIcon className="h-5 w-5" /> },
    { id: "Text", label: "Text", icon: <Type className="h-5 w-5" /> },
    { id: "Work", label: "Work sym.", icon: <NotebookPen className="h-5 w-5" /> },
    { id: "Traffic", label: "Traffic light", icon: <TrafficCone className="h-5 w-5" /> },
    { id: "Timetable", label: "Timetable", icon: <Calendar className="h-5 w-5" /> },
    { id: "Timer", label: "Timer", icon: <TimerIcon className="h-5 w-5" /> },
    { id: "Clock", label: "Clock", icon: <Clock3 className="h-5 w-5" /> },
  ];

  // Keyboard shortcut: Cmd/Ctrl+K to open search
  useEffect(() => {
    const onKey = (e) => {
      const isModK = (e.key.toLowerCase() === "k") && (e.metaKey || e.ctrlKey);
      if (isModK) {
        e.preventDefault();
        setOpen(true);
        setTimeout(() => inputRef.current?.focus(), 0);
      }
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, []);

  const filtered = widgets.filter(w => w.label.toLowerCase().includes(query.toLowerCase()));

  function launch(id) {
    // For mockup: just record recents
    setRecent((prev) => [id, ...prev.filter(x => x !== id)].slice(0, 5));
    setOpen(false);
  }

  return (
    <>
      {/* Collapsed pill */}
      <div className="pointer-events-auto fixed bottom-4 left-1/2 -translate-x-1/2 z-40">
        <div className="rounded-full bg-white/70 dark:bg-slate-900/60 backdrop-blur-xl border border-white/40 dark:border-white/10 shadow-[0_8px_24px_rgba(0,0,0,0.28)] px-3 py-2 flex items-center gap-2">
          {favorites.map((id) => {
            const w = widgets.find((x) => x.id === id);
            if (!w) return null;
            return (
              <button
                key={id}
                className="h-9 w-9 grid place-items-center rounded-full hover:bg-white/80 dark:hover:bg-white/10 transition"
                title={w.label}
                onClick={() => launch(id)}
              >
                {w.icon}
              </button>
            );
          })}
          <div className="mx-1 h-6 w-px bg-black/10 dark:bg-white/10" />
          <button
            className="h-9 px-3 rounded-full bg-white/80 dark:bg-white/10 hover:bg-white/90 dark:hover:bg-white/20 text-sm inline-flex items-center gap-2 border border-white/40 dark:border-white/10"
            onClick={() => setOpen(true)}
            aria-label="Öppna launcher"
          >
            <MoreHorizontal className="h-4 w-4" /> More
          </button>
        </div>
      </div>

      {/* Expanded launcher */}
      {open && (
        <div className="fixed inset-0 z-50">
          <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm" onClick={() => setOpen(false)} />
          <div className="absolute left-1/2 -translate-x-1/2 bottom-5 w-[min(960px,92vw)]">
            <div className="rounded-2xl bg-white/90 dark:bg-slate-900/90 backdrop-blur-xl border border-white/40 dark:border-white/10 shadow-[0_30px_80px_rgba(0,0,0,0.45)] p-4">
              {/* Header/Search */}
              <div className="flex items-center gap-3">
                <div className="relative flex-1">
                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-500" />
                  <input
                    ref={inputRef}
                    placeholder="Sök widgets…  (Ctrl/Cmd + K)"
                    value={query}
                    onChange={(e) => setQuery(e.target.value)}
                    className="w-full pl-9 pr-3 h-11 rounded-xl bg-white text-slate-900 ring-1 ring-black/10 focus:ring-2 focus:ring-indigo-500 outline-none dark:bg-slate-800 dark:text-white dark:ring-white/10"
                  />
                </div>
                <button
                  className="h-11 px-3 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white inline-flex items-center gap-2"
                  onClick={() => setOpen(false)}
                >
                  Stäng
                </button>
              </div>

              {/* Recents & Favorites */}
              <div className="mt-4 grid grid-cols-1 md:grid-cols-3 gap-4">
                <Section title="Snabbval (favoriter)">
                  <div className="flex flex-wrap gap-2">
                    {favorites.map((id) => (
                      <Chip key={id} label={id} onClick={() => launch(id)} />
                    ))}
                  </div>
                </Section>
                <Section title="Senaste">
                  <div className="flex flex-wrap gap-2">
                    {recent.length === 0 ? (
                      <p className="text-sm text-slate-500">— Inget ännu</p>
                    ) : (
                      recent.map((id) => <Chip key={id} label={id} onClick={() => launch(id)} />)
                    )}
                  </div>
                </Section>
                <Section title="Lägg till">
                  <div className="flex flex-wrap gap-2">
                    <button
                      className="h-9 px-3 rounded-xl bg-white/80 dark:bg-white/10 hover:bg-white/90 dark:hover:bg-white/20 border border-white/40 dark:border-white/10 inline-flex items-center gap-2"
                      onClick={() => alert("Mock: Lägg till favorit")}
                    >
                      <Plus className="h-4 w-4" /> Favorit
                    </button>
                  </div>
                </Section>
              </div>

              {/* Grid of results */}
              <div className="mt-4 grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3">
                {filtered.map((w) => (
                  <button
                    key={w.id}
                    onClick={() => launch(w.id)}
                    className="group rounded-xl border border-black/10 dark:border-white/10 bg-white/80 dark:bg-white/5 hover:bg-white/100 dark:hover:bg-white/10 p-3 text-left transition"
                  >
                    <div className="h-9 w-9 rounded-lg grid place-items-center bg-indigo-50 text-indigo-600 dark:bg-indigo-950/40 dark:text-indigo-300 mb-2">
                      {w.icon}
                    </div>
                    <div className="font-medium text-slate-900 dark:text-slate-100">{w.label}</div>
                    <div className="text-xs text-slate-500">Klicka för att lägga till</div>
                  </button>
                ))}
              </div>

              <p className="mt-3 text-xs text-slate-500">
                Tips: Docken visar bara favoritikoner. Tryck <kbd className="px-1 py-0.5 rounded bg-black/10">Ctrl/Cmd+K</kbd> för supersnabb sökning.
              </p>
            </div>
          </div>
        </div>
      )}
    </>
  );
}

function Section({ title, children }) {
  return (
    <div>
      <div className="text-xs uppercase tracking-wide text-slate-500 mb-2">{title}</div>
      {children}
    </div>
  );
}

function Chip({ label, onClick }) {
  return (
    <button
      onClick={onClick}
      className="h-8 px-3 rounded-full bg-white/80 dark:bg-white/10 hover:bg-white/90 dark:hover:bg-white/20 border border-white/40 dark:border-white/10 text-sm"
    >
      {label}
    </button>
  );
}
